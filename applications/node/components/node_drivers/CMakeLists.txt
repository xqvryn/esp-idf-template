set(sources "src/anchor.cpp")
if(COMMAND idf_component_register)
    idf_component_register(SRCS ${sources} INCLUDE_DIRS "include" REQUIRES application)
else()
    add_library(node_drivers STATIC ${sources})
    target_include_directories(node_drivers PUBLIC include)
    target_link_libraries(node_drivers PUBLIC application)
endif()
if(COMMAND idf_component_register)
    set(component_target ${COMPONENT_LIB})
else()
    set(component_target node_drivers)
endif()
target_compile_features(${component_target} PUBLIC cxx_std_20)
set_target_properties(${component_target} PROPERTIES CXX_EXTENSIONS NO)
