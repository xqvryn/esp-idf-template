cmake_minimum_required(VERSION 3.16)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/ProjectVersion.cmake")

if(NOT SDKCONFIG_DEFAULTS AND NOT "$ENV{SDKCONFIG_DEFAULTS}" STREQUAL "")
    set(SDKCONFIG_DEFAULTS "$ENV{SDKCONFIG_DEFAULTS}")
endif()
if(NOT SDKCONFIG_DEFAULTS)
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
endif()
foreach(_local IN ITEMS sdkconfig.defaults.local sdkconfig.local)
    set(_path "${CMAKE_CURRENT_LIST_DIR}/${_local}")
    if(EXISTS "${_path}")
        list(FIND SDKCONFIG_DEFAULTS "${_path}" _idx)
        if(_idx EQUAL -1)
            list(APPEND SDKCONFIG_DEFAULTS "${_path}")
        endif()
    endif()
endforeach()

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(wifi-check)
