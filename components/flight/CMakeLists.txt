set(portable_sources "src/controller.cpp")
if(COMMAND idf_component_register)
    idf_component_register(SRCS ${portable_sources} INCLUDE_DIRS "include" REQUIRES motion parameters)
else()
    add_library(flight STATIC ${portable_sources})
    target_include_directories(flight PUBLIC include)
    target_link_libraries(flight PUBLIC motion parameters)
endif()
if(COMMAND idf_component_register)
    set(component_target ${COMPONENT_LIB})
else()
    set(component_target flight)
endif()
target_compile_features(${component_target} PUBLIC cxx_std_20)
set_target_properties(${component_target} PROPERTIES CXX_EXTENSIONS NO)
