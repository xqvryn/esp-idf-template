set(portable_sources "src/imu.cpp" "src/pwm.cpp" "src/dshot.cpp" "src/rc.cpp" "src/pins.cpp")
if(COMMAND idf_component_register)
    idf_component_register(SRCS ${portable_sources} INCLUDE_DIRS "include")
else()
    add_library(motion STATIC ${portable_sources})
    target_include_directories(motion PUBLIC include)
endif()
if(COMMAND idf_component_register)
    set(component_target ${COMPONENT_LIB})
else()
    set(component_target motion)
endif()
target_compile_features(${component_target} PUBLIC cxx_std_20)
set_target_properties(${component_target} PROPERTIES CXX_EXTENSIONS NO)
