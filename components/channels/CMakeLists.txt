set(portable_sources "src/anchor.cpp")
if(COMMAND idf_component_register)
    idf_component_register(SRCS ${portable_sources} INCLUDE_DIRS "include" REQUIRES runtime)
else()
    add_library(channels STATIC ${portable_sources})
    target_include_directories(channels PUBLIC include)
    target_link_libraries(channels PUBLIC runtime)
endif()
if(COMMAND idf_component_register)
    set(component_target ${COMPONENT_LIB})
else()
    set(component_target channels)
endif()
target_compile_features(${component_target} PUBLIC cxx_std_20)
set_target_properties(${component_target} PROPERTIES CXX_EXTENSIONS NO)
